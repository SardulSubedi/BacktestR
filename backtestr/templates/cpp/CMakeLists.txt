cmake_minimum_required(VERSION 3.20)
project(BacktestRExport LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(backtestr_replay main.cpp)
add_executable(backtestr_core_tests tests/core_tests.cpp)
target_include_directories(backtestr_core_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
foreach(target backtestr_replay backtestr_core_tests)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive-)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
enable_testing()
add_test(NAME core_safety COMMAND backtestr_core_tests)
option(BACKTESTR_ALPACA_PAPER "Build optional Alpaca PAPER broker executable" OFF)
if(BACKTESTR_ALPACA_PAPER)
  find_package(CURL REQUIRED)
  find_package(nlohmann_json 3.10 REQUIRED)
  add_executable(backtestr_paper alpaca_paper.cpp)
  target_link_libraries(backtestr_paper PRIVATE CURL::libcurl nlohmann_json::nlohmann_json)
endif()
